var count = 0;
void Check(bool result, string name)
{
    if (!result) throw new Exception(name);
    Console.WriteLine("PASS " + name);
    count++;
}
bool Valid(BanAnFormViewModel model) =>
    Validator.TryValidateObject(model, new ValidationContext(model), new List<ValidationResult>(), true);
BanAnFormViewModel Form() => new() { TenBan = "Bàn 01", MaLoaiBan = 1, ViTri = "Tầng 1" };
Check(Valid(Form()), "Valid form");
var f = Form(); f.TenBan = "   "; Check(!Valid(f), "Whitespace name");
f = Form(); f.TenBan = new string('a', 101); Check(!Valid(f), "Name length");
f = Form(); f.SoChoNgoi = 0; Check(!Valid(f), "Zero seats");
f = Form(); f.SoChoNgoi = -1; Check(!Valid(f), "Negative seats");
f = Form(); f.MaLoaiBan = 0; Check(!Valid(f), "Missing table type");
f = Form(); f.TrangThai = "invalid"; Check(!Valid(f), "Invalid status");
f = Form(); f.ViTri = ""; Check(!Valid(f), "Missing location");
f = Form(); f.MoTa = null; Check(Valid(f), "Optional description");

var rows = Enumerable.Range(1, 30).Select(i => new BanAn
{
    MaBan = i, TenBan = $"Bàn {i:00}", MaLoaiBan = i <= 20 ? 1 : 2,
    SoChoNgoi = i <= 20 ? 4 : 8, ViTri = i <= 20 ? "Tầng 1" : "Ngoài trời",
    TrangThai = i <= 20 ? BanAnTrangThai.SanSang : BanAnTrangThai.TamNgung
}).AsQueryable();
var filter = new BanAnIndexViewModel
{
    Keyword = "Bàn", MaLoaiBan = 1, SoChoNgoi = 4,
    TrangThai = BanAnTrangThai.SanSang, SortOrder = "name_desc"
};
var query = rows.TraCuu(filter);
Check(query.Count() == 20, "Combined search and three filters");
Check(query.Skip(5).Take(5).Select(x => x.MaBan).SequenceEqual(new[] {15,14,13,12,11}), "Combined sort and second page");
Check(rows.TraCuu(new() { Keyword = "Ngoài trời" }).Count() == 10, "Location search");
Check(!rows.TraCuu(new() { Keyword = "missing" }).Any(), "No search results");
Check(rows.TraCuu(new() { SortOrder = "name_asc" }).First().MaBan == 1, "Ascending name");
Check(rows.TraCuu(new() { SortOrder = "name_desc" }).First().MaBan == 30, "Descending name");
Check(rows.TraCuu(new() { SortOrder = "seat_asc" }).First().SoChoNgoi == 4, "Ascending seats");
Check(rows.TraCuu(new() { SortOrder = "seat_desc" }).First().SoChoNgoi == 8, "Descending seats");
Check(new BanAnIndexViewModel { TotalCount = 0 }.TotalPages == 1, "Empty pagination");
Check(new BanAnIndexViewModel { TotalCount = 21 }.TotalPages == 5, "Partial last page");
var session = new MemorySession();
Check(!Module2AdminAttribute.IsAdmin(session), "Anonymous denied");
session.SetInt32("MaTaiKhoan", 1); session.SetString("VaiTro", "User");
Check(!Module2AdminAttribute.IsAdmin(session), "Customer denied");
session.SetString("VaiTro", "Admin");
Check(Module2AdminAttribute.IsAdmin(session), "Admin allowed");
session.Clear(); session.SetString("VaiTro", "Admin");
Check(!Module2AdminAttribute.IsAdmin(session), "Role without account denied");
Console.WriteLine($"{count} checks passed. Database integration is not covered by these checks.");

sealed class MemorySession : ISession
{
    readonly Dictionary<string, byte[]> values = new();
    public bool IsAvailable => true;
    public string Id => "test";
    public IEnumerable<string> Keys => values.Keys;
    public void Clear() => values.Clear();
    public Task CommitAsync(CancellationToken cancellationToken = default) => Task.CompletedTask;
    public Task LoadAsync(CancellationToken cancellationToken = default) => Task.CompletedTask;
    public void Remove(string key) => values.Remove(key);
    public void Set(string key, byte[] value) => values[key] = value;
    public bool TryGetValue(string key, out byte[] value) => values.TryGetValue(key, out value!);
}
